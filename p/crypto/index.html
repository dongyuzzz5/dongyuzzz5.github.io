#!/usr/bin/env python3  
from Crypto.Cipher import AES
from Crypto.Util.Padding import unpad

KEY_HEX = "00112233445566778899aabbccddeeff"
IV_HEX = "25ef3189e66193cf3e70f74d3f7b18f0"
CIPHER_HEX = "ebd91f0fc53e3d9e2fa5e5ecec1d2ea878a068f6db0c871ac6d14b83bfdb764f"

key = bytes.fromhex(KEY_HEX)
iv = bytes.fromhex(IV_HEX)
ct = bytes.fromhex(CIPHER_HEX)

cipher = AES.new(key, AES.MODE_CBC, iv)
pt_padded = cipher.decrypt(ct)
pt = unpad(pt_padded, 16)

print(pt.decode())

import hashlib
from Crypto.Cipher import AES
from Crypto.Util.Padding import unpad

# 密文（16进制格式）
ciphertext_hex = "5435496e404858cfa2ba08e8ac4f421fd43c289ee54f699da3e403ceff19d8d0"
ciphertext = bytes.fromhex(ciphertext_hex)


# 尝试用 PIN 解密
def decrypt_with_pin(pin):
    # 1. 用 SHA256 对PIN码哈希生成32字节的AES-256密钥
    key = hashlib.sha256(pin.encode()).digest()  # sha256固定输出32字节，刚好满足AES-256的密钥长度要求
    # 2. IV(初始化向量) 是 Key 的前 16 字节
    iv = key[:16]
    # 3. 解密（AES-256-CBC 模式）
    cipher = AES.new(key, AES.MODE_CBC, iv)
    try:
        # 解密后去除PKCS7填充，AES分组长度固定16字节
        plaintext = unpad(cipher.decrypt(ciphertext), AES.block_size)
        return plaintext.decode('utf-8')  # 明文为UTF-8编码的文本
    except (ValueError, UnicodeDecodeError):
        # 解密失败的常见原因：PIN错误、填充错误、明文编码不符
        return None


# 暴力枚举 4位纯数字PIN码 (范围：0000 ~ 9999)
found = False
for pin in range(0, 1000000):
    pin_str = f"{pin:04d}"  # 格式化补零为4位，如 12 → 0012，999 → 0999
    plaintext = decrypt_with_pin(pin_str)
    if plaintext:
        print(f"✅ 成功解密！PIN: {pin_str}, 明文: {plaintext}")
        found = True
        break

if not found:
    print("❌ 未找到匹配的 PIN 爆破范围内")

import base64
from Crypto.Cipher import AES
# 给定的参数
IV = "1229002635654321"  # 初始化向量
KEY = "hellomyaeskeyyyy"  # AES密钥
ENCRYPTED_FLAG = "l60f2FRTj/w6MlIhLWv/rlzEkIhsopz0VcY4W01XvVU="  # Base64编码的密文
def decrypt_aes_cbc(key, iv, ciphertext_b64):
    # Base64解码
    ciphertext = base64.b64decode(ciphertext_b64)
    # 初始化AES-CBC解密器
    cipher = AES.new(key.encode('utf-8'), AES.MODE_CBC, iv.encode('utf-8'))
    # 解密
    plaintext = cipher.decrypt(ciphertext)
    # 去除填充（原代码使用空格填充）
    plaintext = plaintext.decode('utf-8').rstrip(' ')
    return plaintext
# 解密并输出flag
flag = decrypt_aes_cbc(KEY, IV, ENCRYPTED_FLAG)
print("Flag:", flag)

from Crypto.Cipher import AES
from Crypto.Util import Counter
import hashlib

def derive_16byte_key(pin: str, salt: str) -> bytes:
    """生成16字节密钥（取SHA256的前16字节）"""
    full_key = hashlib.sha256((pin + salt).encode()).digest()
    return full_key[:16]  # 取前16字节作为AES-128密钥

def decrypt_ctr(key: bytes, nonce: bytes, ciphertext: bytes) -> bytes:
    """AES-CTR解密（使用16字节密钥）"""
    ctr = Counter.new(64, prefix=nonce, initial_value=0)
    cipher = AES.new(key, AES.MODE_CTR, counter=ctr)
    return cipher.decrypt(ciphertext)

def brute_force_decrypt(ciphertext_hex: str, nonce_hex: str):
    """暴力破解PIN和salt组合"""
    ciphertext = bytes.fromhex(ciphertext_hex)
    nonce = bytes.fromhex(nonce_hex)
    # 完整5位PIN范围 00000-99999，补零格式化
    for pin in range(0, 100000):
        pin_str = f"{pin:05d}"
        for salt in range(0, 100):
            salt_str = f"{salt:02d}"

            # 进度提示
            if pin % 1000 == 0 and salt == 0:
                print(f"尝试中: PIN={pin_str}, Salt={salt_str}")

            # 生成密钥并解密，无多余异常捕获
            key = derive_16byte_key(pin_str, salt_str)
            plaintext = decrypt_ctr(key, nonce, ciphertext)
            if b"flag{" in plaintext:
                print("\n✅ 解密成功！")
                print(f"PIN: {pin_str}, Salt: {salt_str}")
                print(f"明文: {plaintext.decode('utf-8')}")
                return
    print("❌ 未找到正确组合")

if __name__ == "__main__":
    ciphertext_hex = "ad49684af40fb9d2372d32e0df4c8fc1afc843af873f36e79f98c57a81"
    nonce_hex = "6874f1abe89d9025"
    print("开始暴力破解...")
    brute_force_decrypt(ciphertext_hex, nonce_hex)

from hashlib import md5
from Crypto.Cipher import AES
from Crypto.Util.Padding import unpad
from math import isqrt

# 给定参数
p = 115792089237316195423570985008687907853269984665640564039457584007913129639747
Q_components = (123456789, 987654321, 135792468, 864297531)
R_components = (
    53580504271939954579696282638160058429308301927753139543147605882574336327145,
    79991318245209837622945719467562796951137605212294979976479199793453962090891,
    53126869889181040587037210462276116096032594677560145306269148156034757160128,
    97368024230306399859522783292246509699830254294649668434604971213496467857155
)
ciphertext = b'(\xe4IJ\xfd4%\xcf\xad\xb4\x7fi\xae\xdbZux6-\xf4\xd72\x14BB\x1e\xdc\xb7\xb7\xd1\xad#e@\x17\x1f\x12\xc4\xe5\xa6\x10\x91\x08\xd6\x87\x82H\x9e'

# 计算四元数Q的范数
a_q, b_q, c_q, d_q = Q_components
N_Q = (a_q ** 2 + b_q ** 2 + c_q ** 2 + d_q ** 2) % p
print("N(Q) =", N_Q)

# 计算四元数R的范数
a_r, b_r, c_r, d_r = R_components
N_R = (a_r ** 2 + b_r ** 2 + c_r ** 2 + d_r ** 2) % p
print("N(R) =", N_R)


# BSGS算法求解离散对数
def bsgs(g, h, p, bound):
    m = int(isqrt(bound)) + 1  # 步长m
    table = {}  # 存储Baby Steps
    current = 1
    for i in range(m):
        table[current] = i  # 存储g^i mod p和指数i
        current = (current * g) % p
    # 计算Giant Step因子: g^{-m} mod p
    inv_g_m = pow(g, -m, p)
    current = h
    for j in range(m):
        if current in table:
            return j * m + table[current]  # 找到x = j*m + i
        current = (current * inv_g_m) % p
    return None  # 未找到解


bound = 2 ** 50  # secret的上界
secret = bsgs(N_Q, N_R, p, bound)
print("secret =", secret)

# 使用secret生成AES密钥并解密
key = md5(str(secret).encode()).hexdigest().encode()
cipher = AES.new(key, AES.MODE_ECB)
plaintext = unpad(cipher.decrypt(ciphertext), 16)
print("flag =", plaintext.decode())
